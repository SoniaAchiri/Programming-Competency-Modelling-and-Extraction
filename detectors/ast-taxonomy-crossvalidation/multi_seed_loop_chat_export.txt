Here's the caching + multi-seed loop. To avoid re-embedding all 526
files N times (expensive and pointless — the embeddings themselves don't
depend on the split), I embed everything once upfront, then just
re-shuffle/re-split/re-classify per seed using the cached vectors.

Cell A — embed everything once (run this before the loop):

# Pre-embed every file once. Embeddings don't depend on the train/test

# split, so there's no need to recompute them per seed.

embedding_cache = {}

for fname in labeled["file"]:

src = read_src(fname)

if src is not None:

embedding_cache[fname] = codebert_embed(src)

print(f"cached {len(embedding_cache)} embeddings out of {len(labeled)}
files")

Cell B — the multi-seed loop:

import numpy as np

from sklearn.metrics import accuracy_score

SEEDS = list(range(10)) # 10 runs; bump this up if you want tighter
estimates

labels_order = ["manual", "builtin", "novel"]

per_seed_results = []

for seed in SEEDS:

shuffled = labeled.sample(frac=1,
random_state=seed).reset_index(drop=True)

split_point = int(len(shuffled) * 0.7)

train_df_s = shuffled.iloc[:split_point]

test_df_s = shuffled.iloc[split_point:]

# Build centroids from cached embeddings -- no re-embedding

centroids_s = {}

for label in labels_order:

files = train_df_s[train_df_s["structural_label"] ==
label]["file"].tolist()

vecs = [embedding_cache[f] for f in files if f in embedding_cache]

centroids_s[label] = np.mean(vecs, axis=0)

# Classify test set by nearest centroid

rows = []

for _, row in test_df_s.iterrows():

f = row["file"]

if f not in embedding_cache:

continue

emb = embedding_cache[f]

sims = {label: cosine_sim(emb, c) for label, c in centroids_s.items()}

predicted = max(sims, key=sims.get)

rows.append({"file": f, "structural_label": row["structural_label"],
"semantic_predicted": predicted})

results_s = pd.DataFrame(rows)

# Full 3-way accuracy

acc_3way = accuracy_score(results_s["structural_label"],
results_s["semantic_predicted"])

# Manual/builtin-only accuracy -- the figure actually reported in the
paper

mb_only = results_s[results_s["structural_label"].isin(["manual",
"builtin"])]

acc_mb = accuracy_score(mb_only["structural_label"],
mb_only["semantic_predicted"])

# Novel recall specifically, since that's the unstable one

novel_rows = results_s[results_s["structural_label"] == "novel"]

novel_recall = (novel_rows["semantic_predicted"] == "novel").mean() if
len(novel_rows) else float("nan")

per_seed_results.append({

"seed": seed,

"acc_3way": acc_3way,

"acc_manual_builtin": acc_mb,

"novel_recall": novel_recall,

"n_test": len(results_s),

"n_novel_test": len(novel_rows),

})

summary_df = pd.DataFrame(per_seed_results)

print(summary_df)

print()

print(f"3-way accuracy: mean={summary_df['acc_3way'].mean():.3f}
std={summary_df['acc_3way'].std():.3f}")

print(f"manual/builtin acc:
mean={summary_df['acc_manual_builtin'].mean():.3f}
